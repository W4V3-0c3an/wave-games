        (function() {
            // ---------- W4V3 DYNAMIC ACCENT SYSTEM ----------
            // Detects local time of day and drives every accent-colored surface
            // in the stylesheet above via the --accent custom property.
            const W4V3_ACCENTS = {
                morning: '#FFA94D',
                afternoon: '#38D9C9',
                evening: '#FF6B6B',
                night: '#5C8DFF',
                lateNight: '#8B7CFF',
            };
            function w4v3GetPeriod(hour) {
                if (hour < 5) return 'lateNight';
                if (hour < 12) return 'morning';
                if (hour < 17) return 'afternoon';
                if (hour < 21) return 'evening';
                return 'night';
            }
            function w4v3ApplyAccent() {
                const period = w4v3GetPeriod(new Date().getHours());
                document.documentElement.style.setProperty('--accent', W4V3_ACCENTS[period]);
            }
            w4v3ApplyAccent();
            setInterval(w4v3ApplyAccent, 60000); // re-check every minute to catch period boundaries


            // ---------- LOADING STATE ----------
            const body = document.body;
            const introOverlay = document.getElementById('introOverlay');
            const introLogo = document.getElementById('introLogo');
            const loadingSpinner = document.getElementById('loadingSpinner');
            let gamesLoaded = false;
            let labelsLoaded = false;
            
            loadingSpinner.classList.add('show');
            body.classList.add('intro-active');
            
            function checkAllLoaded() {
                if (gamesLoaded && labelsLoaded) {
                    setTimeout(() => {
                        loadingSpinner.classList.remove('show');
                        introLogo.classList.add('animate');
                        body.classList.add('loaded');
                        setTimeout(() => {
                            body.classList.remove('intro-active');
                            introOverlay.classList.add('hidden');
                        }, 1500);
                    }, 100);
                }
            }


            // ---------- PINNED / RECENT (localStorage) ----------
            let pinnedGames = [];
            let recentGames = [];
            let recentExpanded = false;


            function loadPinned() {
                try { const stored = localStorage.getItem('pinnedGames'); pinnedGames = stored ? JSON.parse(stored) : []; } catch { pinnedGames = []; }
            }
            function loadRecent() {
                try { const stored = localStorage.getItem('recentGames'); recentGames = stored ? JSON.parse(stored) : []; } catch { recentGames = []; }
            }
            function savePinned() { localStorage.setItem('pinnedGames', JSON.stringify(pinnedGames)); }
            function saveRecent() { localStorage.setItem('recentGames', JSON.stringify(recentGames)); }
            loadPinned(); loadRecent();

            let allGames = []; // { file, label }

            // ---------- HELPER: play game (fetch from CDN) ----------
            function playGame(file) {
                const normalized = file.includes('.') ? file : file + '.html';
                const encoded = encodeURIComponent(normalized);
                fetch(`https://cdn.jsdelivr.net/gh/bubbls/ugs-singlefile/UGS-Files/${encoded}?t=${Date.now()}`)
                    .then(r => r.text())
                    .then(text => {
                        const newWin = window.open("about:blank", "_blank");
                        if (newWin) { newWin.document.open(); newWin.document.write(text); newWin.document.close(); }
                    });
            }


            // ---------- RECENT & PIN LOGIC ----------
            function addToRecent(file, label) {
                recentGames = recentGames.filter(g => g.file !== file);
                recentGames.unshift({ file, label });
                if (recentGames.length > 50) recentGames.pop();
                saveRecent();
                renderRecentSection();
            }


            function togglePin(file, label) {
                const index = pinnedGames.findIndex(p => p.file === file);
                if (index === -1) pinnedGames.unshift({ file, label });
                else pinnedGames.splice(index, 1);
                savePinned();
                renderPinnedSection();
                updatePinIconsInLetterSections(); // quick icon update in letter cards
            }


            function updatePinIconsInLetterSections() {
                document.querySelectorAll('#letter-sections-container .game-card').forEach(card => {
                    const pinBtn = card.querySelector('.pin-btn');
                    if (!pinBtn) return;
                    const file = card.dataset.file;
                    const isPinned = pinnedGames.some(p => p.file === file);
                    pinBtn.innerHTML = isPinned ? '<i class="fas fa-trash-can"></i>' : '<i class="fas fa-thumbtack"></i>';
                });
            }


            // ---------- FILTER HELPER (applies search to any array of games) ----------
            function filterGamesBySearch(gamesArray) {
                if (!currentSearchTerm.trim()) return gamesArray;
                const term = currentSearchTerm.toLowerCase();
                return gamesArray.filter(g => g.label.toLowerCase().includes(term));
            }


            // ---------- RENDER PINNED SECTION ----------
            function renderPinnedSection() {
                const filteredPinned = filterGamesBySearch(pinnedGames);
                let html = '';
                if (filteredPinned.length === 0 && pinnedGames.length > 0) {
                    // No pinned match search, but we still show the section with a message? Original hides section if no games after filter.
                    // We'll hide the entire section if no matches.
                    pinnedContainer.innerHTML = '';
                    return;
                }
                if (pinnedGames.length === 0) {
                    pinnedContainer.innerHTML = `
                        <div class="top-section pinned-section">
                            <div class="top-header"><i class="fas fa-thumbtack"></i> Pinned Games <span style="font-size:0.8rem; margin-left:auto; color:rgba(255,255,255,0.5);">0</span></div>
                            <div class="games-grid"><div class="no-games-message">📌 No pinned games yet — hover over any game and click the pin.</div></div>
                        </div>
                    `;
                    return;
                }
                let gridHtml = '';
                filteredPinned.forEach((game, index) => {
                    gridHtml += `
                        <div class="game-card" data-file="${game.file}" data-label="${game.label}" style="flex-direction:column; text-align:center;">
                            <span class="game-name">${game.label}</span>
                            <div class="pinned-controls">
                                <button class="move-left" title="move left"><i class="fas fa-chevron-left"></i></button>
                                <button class="move-right" title="move right"><i class="fas fa-chevron-right"></i></button>
                                <button class="unpin-btn" title="unpin"><i class="fas fa-trash-can"></i></button>
                            </div>
                        </div>
                    `;
                });
                html = `
                    <div class="top-section pinned-section">
                        <div class="top-header"><i class="fas fa-thumbtack"></i> Pinned Games <span style="font-size:0.8rem; margin-left:auto; color:rgba(255,255,255,0.5);">${filteredPinned.length}</span></div>
                        <div class="games-grid">${gridHtml}</div>
                    </div>
                `;
                pinnedContainer.innerHTML = html;


                // Attach event listeners
                pinnedContainer.querySelectorAll('.game-card').forEach((card, idx) => {
                    const file = card.dataset.file;
                    const label = card.dataset.label;
                    // Find the actual index in filteredPinned to get correct global index for move operations
                    const globalIndex = pinnedGames.findIndex(p => p.file === file);
                    
                    card.querySelector('.move-left').addEventListener('click', (e) => {
                        e.stopPropagation();
                        if (globalIndex > 0) {
                            [pinnedGames[globalIndex-1], pinnedGames[globalIndex]] = [pinnedGames[globalIndex], pinnedGames[globalIndex-1]];
                            savePinned();
                            renderPinnedSection();
                            updatePinIconsInLetterSections();
                        }
                    });
                    card.querySelector('.move-right').addEventListener('click', (e) => {
                        e.stopPropagation();
                        if (globalIndex < pinnedGames.length-1) {
                            [pinnedGames[globalIndex], pinnedGames[globalIndex+1]] = [pinnedGames[globalIndex+1], pinnedGames[globalIndex]];
                            savePinned();
                            renderPinnedSection();
                            updatePinIconsInLetterSections();
                        }
                    });
                    card.querySelector('.unpin-btn').addEventListener('click', (e) => {
                        e.stopPropagation();
                        pinnedGames = pinnedGames.filter(p => p.file !== file);
                        savePinned();
                        renderPinnedSection();
                        updatePinIconsInLetterSections();
                    });
                    card.addEventListener('click', (e) => {
                        if (e.target.closest('button')) return;
                        playGame(file);
                        addToRecent(file, label);
                    });
                });
            }


            // ---------- RENDER RECENT SECTION ----------
            function renderRecentSection() {
                const filteredRecent = filterGamesBySearch(recentGames);
                if (filteredRecent.length === 0 && recentGames.length > 0) {
                    recentContainer.innerHTML = '';
                    return;
                }
                if (recentGames.length === 0) {
                    recentContainer.innerHTML = `
                        <div class="top-section">
                            <div class="top-header"><i class="fas fa-clock"></i> Recent Games <span style="font-size:0.8rem; margin-left:auto; color:rgba(255,255,255,0.5);">0</span></div>
                            <div class="games-grid"><div class="no-games-message">🕒 No recent games — click any game to start playing.</div></div>
                        </div>
                    `;
                    return;
                }
                const displayGames = recentExpanded ? filteredRecent : filteredRecent.slice(0, 10);
                let gridHtml = '';
                displayGames.forEach(game => {
                    gridHtml += `
                        <div class="game-card" data-file="${game.file}" data-label="${game.label}">
                            <i class="fas fa-gamepad"></i>
                            <span class="game-name">${game.label}</span>
                            <button class="delete-recent-btn" title="remove from recent"><i class="fas fa-times"></i></button>
                        </div>
                    `;
                });
                let footerHtml = '';
                if (recentGames.length > 10) {
                    footerHtml = `<div style="margin-top:16px; text-align:center;"><span class="recent-expand-btn" id="expandRecentBtn">${recentExpanded ? '🔼 Show less' : '🔽 Show all'}</span></div>`;
                }
                recentContainer.innerHTML = `
                    <div class="top-section">
                        <div class="top-header"><i class="fas fa-clock"></i> Recent Games <span style="font-size:0.8rem; margin-left:auto; color:rgba(255,255,255,0.5);">${recentGames.length}</span></div>
                        <div class="games-grid">${gridHtml}</div>
                        ${footerHtml}
                    </div>
                `;


                // Attach listeners
                recentContainer.querySelectorAll('.game-card').forEach(card => {
                    const file = card.dataset.file;
                    const label = card.dataset.label;
                    card.querySelector('.delete-recent-btn').addEventListener('click', (e) => {
                        e.stopPropagation();
                        recentGames = recentGames.filter(r => r.file !== file);
                        saveRecent();
                        renderRecentSection();
                    });
                    card.addEventListener('click', (e) => {
                        if (e.target.closest('.delete-recent-btn')) return;
                        playGame(file);
                        addToRecent(file, label);
                    });
                });
                const expandBtn = document.getElementById('expandRecentBtn');
                if (expandBtn) {
                    expandBtn.addEventListener('click', () => {
                        recentExpanded = !recentExpanded;
                        renderRecentSection();
                    });
                }
            }


            // ---------- RENDER LETTER SECTIONS (from allGames) ----------
            function renderLetterSections() {
                if (!allGames.length) return;
                const filteredAll = filterGamesBySearch(allGames);
                if (filteredAll.length === 0) {
                    lettersContainer.innerHTML = `<div class="no-games-message">✨ No games match your search</div>`;
                    updateLetterNav([]);
                    return;
                }


                // Group by first letter
                const groups = {};
                filteredAll.forEach(game => {
                    const letter = game.label[0].toUpperCase();
                    if (!groups[letter]) groups[letter] = [];
                    groups[letter].push(game);
                });
                const sortedLetters = Object.keys(groups).sort((a,b) => a.localeCompare(b));
                let html = '';
                sortedLetters.forEach(letter => {
                    html += `<div class="letter-section" data-letter="${letter}">`;
                    html += `<div class="letter-header"><span>${letter}</span><hr></div>`;
                    html += `<div class="games-grid">`;
                    groups[letter].forEach(game => {
                        const isPinned = pinnedGames.some(p => p.file === game.file);
                        const pinIcon = isPinned ? 'fa-trash-can' : 'fa-thumbtack';
                        html += `
                            <div class="game-card" data-file="${game.file}" data-label="${game.label}">
                                <i class="fas fa-gamepad"></i>
                                <span class="game-name">${game.label}</span>
                                <button class="pin-btn" title="pin/unpin"><i class="fas ${pinIcon}"></i></button>
                            </div>
                        `;
                    });
                    html += `</div></div>`;
                });
                lettersContainer.innerHTML = html;


                // Attach listeners
                lettersContainer.querySelectorAll('.game-card').forEach(card => {
                    const file = card.dataset.file;
                    const label = card.dataset.label;
                    const pinBtn = card.querySelector('.pin-btn');
                    pinBtn.addEventListener('click', (e) => {
                        e.stopPropagation();
                        togglePin(file, label);
                    });
                    card.addEventListener('click', (e) => {
                        if (e.target.closest('.pin-btn')) return;
                        playGame(file);
                        addToRecent(file, label);
                    });
                });


                updateLetterNav(sortedLetters);
            }


            // ---------- UPDATE LETTER NAVIGATION ----------
            function updateLetterNav(lettersPresent) {
                const alphabet = '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ'.split('');
                letterNav.innerHTML = '';
                alphabet.forEach(letter => {
                    const hasGames = lettersPresent.includes(letter);
                    const btn = document.createElement('div');
                    btn.className = `letter-btn ${hasGames ? '' : 'empty-letter'}`;
                    btn.textContent = letter;
                    if (hasGames) {
                        btn.setAttribute('data-letter', letter);
                        btn.addEventListener('click', () => {
                            const section = document.querySelector(`.letter-section[data-letter="${letter}"]`);
                            if (section) section.scrollIntoView({ behavior: 'smooth', block: 'start' });
                        });
                    }
                    letterNav.appendChild(btn);
                });
            }


            // ---------- FULL RENDER (triggered after data load and on search) ----------
            function renderAll() {
                renderPinnedSection();
                renderRecentSection();
                renderLetterSections();
            }


            // ---------- SCRIPT LOADING (games.js → label handler) ----------
            function loadGamesAndLabels() {
                statusEl.textContent = '📦 loading game stash...';


                // Remove any previously created button sections (from games.js)
                const oldSections = document.querySelectorAll('.letter-section');
                oldSections.forEach(s => s.remove());


                // Load games.js first
                const gameScript = document.createElement('script');
                gameScript.src = 'https://cdn.jsdelivr.net/gh/bubbls/ugs-singlefile@main/games.js';
                gameScript.async = true;
                gameScript.onload = () => {
                    // games.js has created the button sections – now load label handler
                    const labelScript = document.createElement('script');
                    labelScript.src = 'https://cdn.jsdelivr.net/gh/Drowsey-Works/Singlefile-modding@main/game-label-handler.js';
                    labelScript.async = true;
                    labelScript.onload = () => {
                        // label handler has populated window.gameLabels from the buttons
                        if (window.gameLabels && Object.keys(window.gameLabels).length > 0) {
                            allGames = Object.keys(window.gameLabels).map(file => ({ file, label: window.gameLabels[file] }));
                            allGames.sort((a,b) => a.label.localeCompare(b.label));
                            // Remove the original button sections (they are no longer needed)
                            const generatedSections = document.querySelectorAll('.letter-section');
                            generatedSections.forEach(s => s.remove());
                            renderAll();
                            statusEl.textContent = '';
                        } else {
                            statusEl.textContent = '⚠️ no game labels found.';
                        }
                        labelsLoaded = true;
                        checkAllLoaded();
                    };
                    labelScript.onerror = () => {
                        statusEl.textContent = '⚠️ failed to load labels.';
                        labelsLoaded = true;
                        checkAllLoaded();
                    };
                    document.body.appendChild(labelScript);
                    gamesLoaded = true;
                    checkAllLoaded();
                };
                gameScript.onerror = () => {
                    statusEl.textContent = '⚠️ failed to load games.';
                    gamesLoaded = true;
                    labelsLoaded = true;
                    checkAllLoaded();
                    lettersContainer.innerHTML = '<div class="no-games-message">could not load data. refresh?</div>';
                };
                document.body.appendChild(gameScript);
            }


            // Start loading
            loadGamesAndLabels();


            // ---------- SEARCH LISTENER ----------
            let debounceTimer;
            searchInput.addEventListener('input', () => {
                clearTimeout(debounceTimer);
                debounceTimer = setTimeout(() => {
                    currentSearchTerm = searchInput.value;
                    renderAll();
                }, 150);
            });
        })();
    </script>
